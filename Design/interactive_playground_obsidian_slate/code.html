Here is an optimized, non-allocating streaming EMA accumulator implemented with inline primitives:

#[derive(Debug, Clone, Copy)]
pub struct StreamingEma {
    alpha: f64,
    current: Option<f64>,
}

impl StreamingEma {
    pub const fn new(period: usize) -> Self {
        Self { alpha: 2.0 / (period as f64 + 1.0), current: None }
    }

    #[inline(always)]
    pub fn update(&mut self, value: f64) -> f64 {
        let next = match self.current {
            Some(prev) => prev + self.alpha * (value - prev),
            None => value,
        };
        self.current = Some(next);
        next
    }
}
